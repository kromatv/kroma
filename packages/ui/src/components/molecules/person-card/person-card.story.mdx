import { defineStory } from '@kroma/workbench/story';
import { AVATAR_GRADIENTS } from '#ui/components/atoms/avatar';
import { Box } from '#ui/components/atoms/box';
import { PersonCard, personCardVariants } from './person-card';
import { CAST } from './person-card.fixtures';

export const story = defineStory({
  group: 'Media',
  variants: personCardVariants,
  matrix: true,
  component: PersonCard,
  args: { name: 'Sigourney Weaver', role: 'Ellen Ripley' },
});

A person, as a circle: the cast rail tile at both distances. The focus ring is drawn on the **photo**, never as a box around the tile and its caption: a rectangle around a circle reads as a form field, a lit circle reads as a face being pointed at. On glass there is no focus, so the card dims under the thumb instead.

```tsx
<PersonCard name={member.name} role={member.character} photo={art} onPress={open} />
<PersonCard size="sm" name={member.name} role={member.character} />
```

<Scene name="A cast rail">

  {story.take((props) => (
    <Box row gap={24}>
      {CAST.map((member, i) => (
        <PersonCard
          key={member.name}
          {...props}
          name={member.name}
          role={member.role}
          gradient={AVATAR_GRADIENTS[i % AVATAR_GRADIENTS.length]}
        />
      ))}
    </Box>
  ))}
</Scene>
