import { defineStory } from '@kromatv/workbench/story';
import { Box } from '#ui/components/atoms/box';
import { Ground } from '#ui/components/atoms/ground';
import { StageFlashView } from './stage-flash';

export const story = defineStory({
  group: 'Player',
  viewport: 'tv',
  args: { kind: 'seek', deltaSec: 10, level: 1.5, scale: 1 },
  controls: {
    kind: ['seek', 'volume'],
    deltaSec: { min: -600, max: 600, step: 10 },
    level: { min: 0, max: 2, step: 0.05 },
    scale: { min: 0.5, max: 1, step: 0.05 },
  },
  render: ({ kind, deltaSec, level, scale }) => (
    <Ground tone="dark">
      <Box aspect={16 / 9} minH={280} bg="surface2" radius="lg" overflow="hidden">
        <StageFlashView
          scale={scale}
          flash={
            kind === 'seek'
              ? { kind: 'seek', dir: deltaSec < 0 ? -1 : 1, deltaSec }
              : { kind: 'volume', level }
          }
        />
      </Box>
    </Ground>
  ),
});

It fades out on its own once the flash it was given goes null, holding the last label through the fade.
